124 enum StrobeCommand {
125 CC1101_SRES = 0x30, // Reset chip.
126 CC1101_SFSTXON = 0x31, // Enable and calibrate frequency synthesizer (if MCSM0.FS_AUTOCAL=1).
127 // If in RX/TX: Go to a wait state where only the synthesizer is
128 // running (for quick RX / TX turnaround).
129 CC1101_SXOFF = 0x32, // Turn off crystal oscillator.
130 CC1101_SCAL = 0x33, // Calibrate frequency synthesizer and turn it off
131 // (enables quick start).
132 CC1101_SRX = 0x34, // Enable RX. Perform calibration first if coming from IDLE and
133 // MCSM0.FS_AUTOCAL=1.
134 CC1101_STX = 0x35, // In IDLE state: Enable TX. Perform calibration first if
135 // MCSM0.FS_AUTOCAL=1. If in RX state and CCA is enabled:
136 // Only go to TX if channel is clear.
137 CC1101_SIDLE = 0x36, // Exit RX / TX, turn off frequency synthesizer and exit
138 // Wake-On-Radio mode if applicable.
139 CC1101_SAFC = 0x37, // Perform AFC adjustment of the frequency synthesizer
140 CC1101_SWOR = 0x38, // Start automatic RX polling sequence (Wake-on-Radio)
141 CC1101_SPWD = 0x39, // Enter power down mode when CSn goes high.
142 CC1101_SFRX = 0x3A, // Flush the RX FIFO buffer.
143 CC1101_SFTX = 0x3B, // Flush the TX FIFO buffer.
144 CC1101_SWORRST = 0x3C, // Reset real time clock.
145 CC1101_SNOP = 0x3D, // No operation. May be used to pad strobe commands to two
146 // INT8Us for simpler software.
147 };
